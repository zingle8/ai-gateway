自定义 GlobalFilter 示例
Reactor 响应式写法 · 无阻塞
@Component @Order(-100) public class AuthGlobalFilter implements GlobalFilter { @Override public Mono<Void> filter(ServerWebExchange exchange, GatewayFilterChain chain) { ServerHttpRequest req = exchange.getRequest(); String appCode = req.getHeaders() .getFirst("X-App-Code"); // 1. 本地缓存优先，避免每次远程校验 AuthContext ctx = authCache.getIfPresent(appCode); // 2. 缓存未命中 → 响应式远程校验（非阻塞） Mono<AuthContext> auth = ctx != null ? Mono.just(ctx) : authClient.verify(appCode) .doOnNext(c -> authCache.put(appCode, c)); return auth .flatMap(c -> { exchange.getAttributes() .put("AUTH_CTX", c); return chain.filter(exchange); }) .switchIfEmpty(Mono.defer(() -> forbidden(exchange, "401 鉴权失败"))); } }
